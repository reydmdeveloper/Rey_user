# -*- coding: utf-8 -*-
"""
Windows version-information resource for the REYDM executable.

This is what populates the EXE's right-click → Properties → Details tab
(Company name, Product name, File description, etc.) and the publisher string
Windows reads from the binary.

EDIT the values marked >>> EDIT <<< to your real company details, then it is
referenced from REYDM.spec via:  version="version_info.txt"

Note: this sets the *metadata* publisher. The "Unknown Publisher" prompt in
Windows SmartScreen/UAC is separate and only disappears with a real
code-signing certificate (see INSTALL_BUILD_GUIDE.md).
"""

VSVersionInfo(
    ffi=FixedFileInfo(
        # filevers / prodvers: (major, minor, patch, build)
        filevers=(1, 0, 0, 0),
        prodvers=(1, 0, 0, 0),
        mask=0x3F,
        flags=0x0,
        OS=0x40004,
        fileType=0x1,
        subtype=0x0,
        date=(0, 0),
    ),
    kids=[
        StringFileInfo(
            [
                StringTable(
                    "040904B0",  # U.S. English, Unicode
                    [
                        # >>> EDIT <<< your company / publisher name
                        StringStruct("CompanyName", "REY Datamind"),
                        StringStruct("FileDescription", "REYDM — REY Datamind Platform"),
                        StringStruct("FileVersion", "1.0.0.0"),
                        StringStruct("InternalName", "REYDM"),
                        StringStruct("LegalCopyright", "© 2026 REY Datamind. All rights reserved."),
                        StringStruct("OriginalFilename", "REYDM.exe"),
                        StringStruct("ProductName", "REYDM"),
                        StringStruct("ProductVersion", "1.0.0.0"),
                    ],
                )
            ]
        ),
        VarFileInfo([VarStruct("Translation", [0x0409, 1200])]),
    ],
)
